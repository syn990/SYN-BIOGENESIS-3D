# The book of words for species: what the SYN edition says a being is.
#
# The format (src/syn/lexicon.h): a [bank] has an id and words=
# lines of phrases split on |; a phrase may name other banks in {braces}.
# An id may carry a tag, the being's kind (src/syn/kind.c): a lookup for
# `habit` on a grazer takes habit.grazer, else plain habit. "~" is an
# empty phrase.
#
# Bound by the program for each code, never written here:
#   {size} {Size}  tiny, small, mid-sized, large, huge (the full-grown body)
#   {arms}         one-armed ... eight-armed
#   {mirror}       "mirrored, " or nothing
#   {genes}        "one gene", "12 genes"
#   {main}         its main colours: "mostly C4", "green and blue"
#   {kind}         its kind, in lower case: "grazer"
#
# What a kind does here must be true of every being of that kind: the
# kinds are read from the colours, and the colours' rules are the game's.

[bank]
id=species
words={Size}, {mirror}{arms} {kind}, {main}. {habit} {closing}
words=A {size}, {mirror}{arms} {kind}: {main}, {genes}. {habit}
words={Size} and {arms}, {main}. {habit} {closing}
words=Survey notes have it as a {size} {kind}: {mirror}{arms}, {main}. {habit}
words={Size} {kind}, {arms}, {genes}, {main}. {habit} {closing}

[bank]
id=closing
